#include <ESP8266WiFi.h>
#include <ESP8266WebServer.h>
#include <EEPROM.h>

// Configuration
const char* AP_SSID = "DCN";
const char* AP_PASSWORD = "12345678";
const int MAX_MESSAGES = 50;
const int MESSAGE_LENGTH = 200;
const int USERNAME_LENGTH = 32;
const int EEPROM_SIZE = 4096;
const int LED_PIN = 14; // Built-in LED pin (can be changed to any GPIO pin)

// Web server
ESP8266WebServer server(80);

// Message structure (updated to include username)
struct Message {
  char content[MESSAGE_LENGTH];
  char username[USERNAME_LENGTH];
  unsigned long timestamp;
  bool valid;
};

// Global variables
Message messages[MAX_MESSAGES];
int messageCount = 0;
int currentIndex = 0;
bool ledState = false;

void setup() {
  Serial.begin(115200);
  delay(1000);
  
  // Initialize LED pin
  pinMode(LED_PIN, OUTPUT);
  digitalWrite(LED_PIN, LOW); // Turn off LED initially
  
  // Initialize EEPROM
  EEPROM.begin(EEPROM_SIZE);
  
  // Load messages from EEPROM
  loadMessages();
  
  // Setup WiFi Access Point
  WiFi.mode(WIFI_AP);
  WiFi.softAP(AP_SSID, AP_PASSWORD);
  
  Serial.println("Access Point Started");
  Serial.print("IP Address: ");
  Serial.println(WiFi.softAPIP());
  
  // Setup web server routes
  server.on("/", handleRoot);
  server.on("/send", HTTP_POST, handleSendMessage);
  server.on("/messages", HTTP_GET, handleGetMessages);
  server.on("/clear", HTTP_POST, handleClearMessages);
  server.on("/led", HTTP_POST, handleLED);
  server.on("/ledstatus", HTTP_GET, handleLEDStatus);
  server.on("/style.css", handleCSS);
  server.on("/script.js", handleJS);
  server.onNotFound(handleNotFound);
  
  server.begin();
  Serial.println("HTTP server started");
}

void loop() {
  server.handleClient();
  delay(10); // Small delay for stability
}

void loadMessages() {
  int addr = 0;
  
  // Read message count
  EEPROM.get(addr, messageCount);
  addr += sizeof(int);
  
  // Read current index
  EEPROM.get(addr, currentIndex);
  addr += sizeof(int);
  
  // Read LED state
  EEPROM.get(addr, ledState);
  addr += sizeof(bool);
  
  // Apply LED state
  digitalWrite(LED_PIN, ledState ? HIGH : LOW);
  
  // Validate data
  if (messageCount < 0 || messageCount > MAX_MESSAGES || 
      currentIndex < 0 || currentIndex >= MAX_MESSAGES) {
    messageCount = 0;
    currentIndex = 0;
    return;
  }
  
  // Read messages
  for (int i = 0; i < MAX_MESSAGES; i++) {
    EEPROM.get(addr, messages[i]);
    addr += sizeof(Message);
  }
}

void saveMessages() {
  int addr = 0;
  
  // Save message count
  EEPROM.put(addr, messageCount);
  addr += sizeof(int);
  
  // Save current index
  EEPROM.put(addr, currentIndex);
  addr += sizeof(int);
  
  // Save LED state
  EEPROM.put(addr, ledState);
  addr += sizeof(bool);
  
  // Save messages
  for (int i = 0; i < MAX_MESSAGES; i++) {
    EEPROM.put(addr, messages[i]);
    addr += sizeof(Message);
  }
  
  EEPROM.commit();
}

void addMessage(const String& content, const String& username) {
  // Clear the message slot
  memset(&messages[currentIndex], 0, sizeof(Message));
  
  // Copy content (truncate if too long)
  strncpy(messages[currentIndex].content, content.c_str(), MESSAGE_LENGTH - 1);
  messages[currentIndex].content[MESSAGE_LENGTH - 1] = '\0';
  
  // Copy username (truncate if too long)
  strncpy(messages[currentIndex].username, username.c_str(), USERNAME_LENGTH - 1);
  messages[currentIndex].username[USERNAME_LENGTH - 1] = '\0';
  
  messages[currentIndex].timestamp = millis();
  messages[currentIndex].valid = true;
  
  // Update counters
  currentIndex = (currentIndex + 1) % MAX_MESSAGES;
  if (messageCount < MAX_MESSAGES) {
    messageCount++;
  }
  
  // Save to EEPROM
  saveMessages();
}

void clearAllMessages() {
  messageCount = 0;
  currentIndex = 0;
  
  // Clear all messages
  for (int i = 0; i < MAX_MESSAGES; i++) {
    memset(&messages[i], 0, sizeof(Message));
    messages[i].valid = false;
  }
  
  saveMessages();
}

String escapeJsonString(const String& str) {
  String escaped = "";
  for (int i = 0; i < str.length(); i++) {
    char c = str.charAt(i);
    switch (c) {
      case '"':  escaped += "\\\""; break;
      case '\\': escaped += "\\\\"; break;
      case '\b': escaped += "\\b"; break;
      case '\f': escaped += "\\f"; break;
      case '\n': escaped += "\\n"; break;
      case '\r': escaped += "\\r"; break;
      case '\t': escaped += "\\t"; break;
      default:   escaped += c; break;
    }
  }
  return escaped;
}

String getMessagesJSON() {
  String json = "[";
  
  if (messageCount == 0) {
    json += "]";
    return json;
  }
  
  // Calculate starting index for circular buffer
  int startIdx = (messageCount == MAX_MESSAGES) ? currentIndex : 0;
  bool first = true;
  
  for (int i = 0; i < messageCount; i++) {
    int idx = (startIdx + i) % MAX_MESSAGES;
    if (messages[idx].valid) {
      if (!first) {
        json += ",";
      }
      json += "{";
      json += "\"content\":\"" + escapeJsonString(String(messages[idx].content)) + "\",";
      json += "\"username\":\"" + escapeJsonString(String(messages[idx].username)) + "\",";
      json += "\"timestamp\":" + String(messages[idx].timestamp);
      json += "}";
      first = false;
    }
  }
  
  json += "]";
  return json;
}

void handleSendMessage() {
  if (!server.hasArg("message") || !server.hasArg("username")) {
    server.send(400, "application/json", "{\"error\":\"Message and username required\"}");
    return;
  }
  
  String message = server.arg("message");
  String username = server.arg("username");
  message.trim();
  username.trim();
  
  if (message.length() == 0 || username.length() == 0) {
    server.send(400, "application/json", "{\"error\":\"Message and username cannot be empty\"}");
    return;
  }
  
  addMessage(message, username);
  server.send(200, "application/json", "{\"success\":true}");
}

void handleGetMessages() {
  String json = getMessagesJSON();
  server.send(200, "application/json", json);
}

void handleClearMessages() {
  clearAllMessages();
  server.send(200, "application/json", "{\"success\":true}");
}

void handleLED() {
  if (!server.hasArg("state")) {
    server.send(400, "application/json", "{\"error\":\"LED state required\"}");
    return;
  }
  
  String state = server.arg("state");
  ledState = (state == "on" || state == "1" || state == "true");
  
  // Apply LED state (normal logic: HIGH = ON, LOW = OFF)
  digitalWrite(LED_PIN, ledState ? HIGH : LOW);
  
  // Save LED state to EEPROM
  saveMessages();
  
  String response = "{\"success\":true,\"state\":\"" + String(ledState ? "on" : "off") + "\"}";
  server.send(200, "application/json", response);
}

void handleLEDStatus() {
  String response = "{\"state\":\"" + String(ledState ? "on" : "off") + "\"}";
  server.send(200, "application/json", response);
}
